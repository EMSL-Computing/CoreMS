  25def hdf5_dataset_kwargs(shape, compression=DEFAULT_HDF5_COMPRESSION, vlen_str=False):
  26    """Build ``h5py.Group.create_dataset`` filter kwargs for a compression choice.
  27
  28    Centralizes the HDF5 writers' compression choice (one helper used by every
  29    ``create_dataset`` call on the mass spectrum / LC-MS export path) and the
  30    filter exceptions h5py enforces: it refuses chunk/filter options on scalar
  31    (``shape == ()``) or zero-length datasets, and the fletcher32 checksum
  32    filter gains nothing (and h5py can reject it) on variable-length string
  33    datasets.
  34
  35    Parameters
  36    ----------
  37    shape : tuple
  38        Shape of the data about to be written.
  39    compression : int, str or None, optional
  40        A gzip level 0-9, ``"lzf"``, or ``None``/``"none"`` for no
  41        compression. Default is 4. ``"lzf"`` is h5py's own filter, not part
  42        of core HDF5 -- a file written with it needs h5py (or another tool
  43        with the lzf plugin available) to read the compressed datasets
  44        back; gzip is the portable choice across other HDF5 tools.
  45    vlen_str : bool, optional
  46        True when the dataset's dtype is a variable-length string
  47        (``h5py.string_dtype()``); skips the shuffle/fletcher32 filters that
  48        don't apply to vlen data. Default is False.
  49
  50    Returns
  51    -------
  52    dict
  53        Keyword arguments to splat into ``create_dataset`` (besides ``data``
  54        and ``dtype``).
  55    """
  56    if not shape or 0 in shape:
  57        return {}
  58
  59    kwargs = {"chunks": True}
  60
  61    if compression not in (None, "none"):
  62        if compression == "lzf":
  63            kwargs["compression"] = "lzf"
  64        else:
  65            kwargs["compression"] = "gzip"
  66            kwargs["compression_opts"] = int(compression)
  67
  68    if not vlen_str:
  69        kwargs["shuffle"] = True
  70        kwargs["fletcher32"] = True
  71
  72    return kwargs

 414    def add_mass_spectrum_to_hdf5(
 415        self,
 416        hdf_handle,
 417        mass_spectrum,
 418        group_key,
 419        mass_spectra_group=None,
 420        export_raw=True,
 421        compression=DEFAULT_HDF5_COMPRESSION,
 422        profile_dtype="f4",
 423    ):
 424        """Adds the mass spectrum data to an HDF5 file.
 425
 426        Parameters
 427        ----------
 428        hdf_handle : h5py.File
 429            The HDF5 file handle.
 430        mass_spectrum : MassSpectrum
 431            The mass spectrum to add to the HDF5 file.
 432        group_key : str
 433            The group key (where to add the mass spectrum data within the HDF5 file).
 434        mass_spectra_group : h5py.Group, optional
 435            The mass spectra group. Defaults to None (no group, mass spectrum is added to the root).
 436        export_raw : bool, optional
 437            Whether to export the raw data. Defaults to True.
 438            If False, only the processed data (peaks) is exported (essentially centroided data).
 439        compression : int, str or None, optional
 440            Compression for the written datasets: a gzip level 0-9, ``"lzf"``,
 441            or ``None``/``"none"``. Default is 4. ``"lzf"`` is h5py's own
 442            filter, not a standard HDF5 one -- it needs h5py (or a tool
 443            with the lzf plugin) to read the result back.
 444        profile_dtype : str, optional
 445            dtype used for the stored profile m/z/abundance arrays
 446            (``"f4"`` or ``"f8"``). Default is ``"f4"``.
 447        """
 448        if mass_spectra_group is None:
 449            # Check if the file has the necessary attributes and add them if not
 450            # This assumes that if there is a mass_spectra_group, these attributes were already added to the file
 451            if not hdf_handle.attrs.get("date_utc"):
 452                timenow = str(
 453                    datetime.now(timezone.utc).strftime("%d/%m/%Y %H:%M:%S %Z")
 454                )
 455                hdf_handle.attrs["date_utc"] = timenow
 456                hdf_handle.attrs["file_name"] = mass_spectrum.filename.name
 457                hdf_handle.attrs["data_structure"] = "mass_spectrum"
 458                hdf_handle.attrs["analyzer"] = mass_spectrum.analyzer
 459                hdf_handle.attrs["instrument_label"] = mass_spectrum.instrument_label
 460                hdf_handle.attrs["sample_name"] = mass_spectrum.sample_name
 461
 462        list_results = self.list_dict_to_list(mass_spectrum, is_hdf5=True)
 463
 464        dict_ms_attrs = self.get_mass_spec_attrs(mass_spectrum)
 465
 466        setting_dicts = parameter_to_dict.get_dict_data_ms(mass_spectrum)
 467
 468        columns_labels = json.dumps(
 469            self.columns_label + self.get_all_used_atoms_in_order(mass_spectrum),
 470            sort_keys=False,
 471            indent=4,
 472            separators=(",", ": "),
 473        )
 474
 475        group_key = group_key
 476
 477        if mass_spectra_group is not None:
 478            hdf_handle = mass_spectra_group
 479
 480        if group_key not in hdf_handle.keys():
 481            scan_group = hdf_handle.create_group(group_key)
 482
 483            # If there is raw data (from profile data) save it
 484            if not mass_spectrum.is_centroid and export_raw:
 485                # "mz" is always the experimental (uncalibrated) profile; a
 486                # separate "mz_cal" row is added only when the spectrum is
 487                # actually calibrated. mz_cal_profile alone isn't enough to
 488                # tell: MzDomainCalibration.__init__ pre-populates it before
 489                # any calibration has run, so a never-calibrated spectrum
 490                # can still have a (stale/placeholder) mz_cal_profile.
 491                row_labels = ["mz", "abundance"]
 492                rows = [
 493                    np.asarray(mass_spectrum._mz_exp),
 494                    np.asarray(mass_spectrum.abundance_profile),
 495                ]
 496                mz_cal_profile = mass_spectrum.mz_cal_profile
 497                if mass_spectrum.is_calibrated and mz_cal_profile is not None:
 498                    row_labels.append("mz_cal")
 499                    rows.append(np.asarray(mz_cal_profile))
 500
 501                mz_abun_array = np.array(rows, dtype=profile_dtype)
 502
 503                raw_ms_dataset = scan_group.create_dataset(
 504                    "raw_ms",
 505                    data=mz_abun_array,
 506                    dtype=profile_dtype,
 507                    **hdf5_dataset_kwargs(mz_abun_array.shape, compression=compression),
 508                )
 509                raw_ms_dataset.attrs["row_labels"] = json.dumps(row_labels)
 510                # Recorded independently of whether an "mz_cal" row was
 511                # written, so a reader can trust this over row presence.
 512                raw_ms_dataset.attrs["is_calibrated"] = bool(
 513                    mass_spectrum.is_calibrated
 514                )
 515
 516                # Add metadata to the raw_ms dataset when it exists
 517                raw_ms_dataset.attrs["MassSpecAttrs"] = json.dumps(dict_ms_attrs)
 518
 519                if isinstance(mass_spectrum, MassSpecfromFreq):
 520                    raw_ms_dataset.attrs["TransientSetting"] = json.dumps(
 521                        setting_dicts.get("TransientSetting"),
 522                        sort_keys=False,
 523                        indent=4,
 524                        separators=(",", ": "),
 525                    )
 526            else:
 527                # For centroided data, store metadata directly on the scan group
 528                scan_group.attrs["MassSpecAttrs"] = json.dumps(dict_ms_attrs)
 529
 530                if isinstance(mass_spectrum, MassSpecfromFreq):
 531                    scan_group.attrs["TransientSetting"] = json.dumps(
 532                        setting_dicts.get("TransientSetting"),
 533                        sort_keys=False,
 534                        indent=4,
 535                        separators=(",", ": "),
 536                    )
 537
 538        else:
 539            scan_group = hdf_handle.get(group_key)
 540
 541        # Next processed-data index: one past the highest existing numeric
 542        # key, or 0 if there are none. "raw_ms" and "peak_profile_indices_*"
 543        # are not processed-data indices (get_time_index_to_pull also only
 544        # considers numeric keys, for the same reason). Using the max rather
 545        # than a count matters for a legacy file that already has "raw_ms"
 546        # and "1" (an old off-by-one: pre-fix code counted "raw_ms" as an
 547        # occupied slot) -- counting numeric keys would give "1" again and
 548        # collide with the existing one; max+1 gives "2".
 549        numeric_keys = [int(k) for k in scan_group.keys() if k.isdigit()]
 550        index_processed_data = str(max(numeric_keys) + 1) if numeric_keys else "0"
 551
 552        # Per-peak (left, apex, right) indices into the profile arrays above,
 553        # in mspeaks order, so a reloaded centroid-with-profile mass
 554        # spectrum doesn't need to re-derive peak boundaries from the peak
 555        # table's row position (which isn't a profile index). Written once
 556        # per peak-table version (same suffix as the processed-data index
 557        # below), since the profile itself (raw_ms) doesn't change across
 558        # reprocessing but the picked peaks do -- except raw_ms itself is
 559        # only ever written on the group's first creation (just above), so
 560        # if *this* mass_spectrum's profile is a different length (e.g. a
 561        # later reprocessing that re-picked over different raw data), these
 562        # indices would point into a profile that was never stored. Skip
 563        # writing them in that case; the reader falls back to a search.
 564        existing_raw_ms = scan_group.get("raw_ms")
 565        if (
 566            not mass_spectrum.is_centroid
 567            and export_raw
 568            and existing_raw_ms is not None
 569            and existing_raw_ms.shape[-1] == len(mass_spectrum.abundance_profile)
 570        ):
 571            peak_indices = np.array(
 572                [
 573                    (
 574                        mspeak.peak_left_index,
 575                        mspeak.peak_apex_index,
 576                        mspeak.peak_right_index,
 577                    )
 578                    for mspeak in mass_spectrum.mspeaks
 579                ],
 580                dtype=np.int32,
 581            ).reshape(-1, 3)
 582            scan_group.create_dataset(
 583                f"peak_profile_indices_{index_processed_data}",
 584                data=peak_indices,
 585                **hdf5_dataset_kwargs(peak_indices.shape, compression=compression),
 586            )
 587
 588        timenow = str(datetime.now(timezone.utc).strftime("%d/%m/%Y %H:%M:%S %Z"))
 589
 590        # Convert list_results to numpy array and optimize data types
 591        array_results = np.array(list_results)
 592
 593        # Apply data type optimization for numeric columns
 594        if array_results.dtype == np.float64:
 595            array_results = array_results.astype(np.float32)
 596        elif array_results.dtype == np.int64:
 597            array_results = array_results.astype(np.int32)
 598
 599        processed_dset = scan_group.create_dataset(
 600            index_processed_data,
 601            data=array_results,
 602            **hdf5_dataset_kwargs(array_results.shape, compression=compression),
 603        )
 604
 605        processed_dset.attrs["date_utc"] = timenow
 606
 607        processed_dset.attrs["ColumnsLabels"] = columns_labels
 608
 609        processed_dset.attrs["MoleculaSearchSetting"] = json.dumps(
 610            setting_dicts.get("MoleculaSearch"),
 611            sort_keys=False,
 612            indent=4,
 613            separators=(",", ": "),
 614        )
 615
 616        processed_dset.attrs["MassSpecPeakSetting"] = json.dumps(
 617            setting_dicts.get("MassSpecPeak"),
 618            sort_keys=False,
 619            indent=4,
 620            separators=(",", ": "),
 621        )
 622

 415    def add_mass_spectrum_to_hdf5(
 416        self,
 417        hdf_handle,
 418        mass_spectrum,
 419        group_key,
 420        mass_spectra_group=None,
 421        export_raw=True,
 422        compression=DEFAULT_HDF5_COMPRESSION,
 423        profile_dtype="f4",
 424    ):
 425        """Adds the mass spectrum data to an HDF5 file.
 426
 427        Parameters
 428        ----------
 429        hdf_handle : h5py.File
 430            The HDF5 file handle.
 431        mass_spectrum : MassSpectrum
 432            The mass spectrum to add to the HDF5 file.
 433        group_key : str
 434            The group key (where to add the mass spectrum data within the HDF5 file).
 435        mass_spectra_group : h5py.Group, optional
 436            The mass spectra group. Defaults to None (no group, mass spectrum is added to the root).
 437        export_raw : bool, optional
 438            Whether to export the raw data. Defaults to True.
 439            If False, only the processed data (peaks) is exported (essentially centroided data).
 440        compression : int, str or None, optional
 441            Compression for the written datasets: a gzip level 0-9, ``"lzf"``,
 442            or ``None``/``"none"``. Default is 4. ``"lzf"`` is h5py's own
 443            filter, not a standard HDF5 one -- it needs h5py (or a tool
 444            with the lzf plugin) to read the result back.
 445        profile_dtype : str, optional
 446            dtype used for the stored profile m/z/abundance arrays
 447            (``"f4"`` or ``"f8"``). Default is ``"f4"``.
 448        """
 449        if mass_spectra_group is None:
 450            # Check if the file has the necessary attributes and add them if not
 451            # This assumes that if there is a mass_spectra_group, these attributes were already added to the file
 452            if not hdf_handle.attrs.get("date_utc"):
 453                timenow = str(
 454                    datetime.now(timezone.utc).strftime("%d/%m/%Y %H:%M:%S %Z")
 455                )
 456                hdf_handle.attrs["date_utc"] = timenow
 457                hdf_handle.attrs["file_name"] = mass_spectrum.filename.name
 458                hdf_handle.attrs["data_structure"] = "mass_spectrum"
 459                hdf_handle.attrs["analyzer"] = mass_spectrum.analyzer
 460                hdf_handle.attrs["instrument_label"] = mass_spectrum.instrument_label
 461                hdf_handle.attrs["sample_name"] = mass_spectrum.sample_name
 462
 463        list_results = self.list_dict_to_list(mass_spectrum, is_hdf5=True)
 464
 465        dict_ms_attrs = self.get_mass_spec_attrs(mass_spectrum)
 466
 467        setting_dicts = parameter_to_dict.get_dict_data_ms(mass_spectrum)
 468
 469        columns_labels = json.dumps(
 470            self.columns_label + self.get_all_used_atoms_in_order(mass_spectrum),
 471            sort_keys=False,
 472            indent=4,
 473            separators=(",", ": "),
 474        )
 475
 476        group_key = group_key
 477
 478        if mass_spectra_group is not None:
 479            hdf_handle = mass_spectra_group
 480
 481        if group_key not in hdf_handle.keys():
 482            scan_group = hdf_handle.create_group(group_key)
 483
 484            # If there is raw data (from profile data) save it
 485            if not mass_spectrum.is_centroid and export_raw:
 486                # "mz" is always the experimental (uncalibrated) profile; a
 487                # separate "mz_cal" row is added only when the spectrum is
 488                # actually calibrated. mz_cal_profile alone isn't enough to
 489                # tell: MzDomainCalibration.__init__ pre-populates it before
 490                # any calibration has run, so a never-calibrated spectrum
 491                # can still have a (stale/placeholder) mz_cal_profile.
 492                row_labels = ["mz", "abundance"]
 493                rows = [
 494                    np.asarray(mass_spectrum._mz_exp),
 495                    np.asarray(mass_spectrum.abundance_profile),
 496                ]
 497                mz_cal_profile = mass_spectrum.mz_cal_profile
 498                if mass_spectrum.is_calibrated and mz_cal_profile is not None:
 499                    row_labels.append("mz_cal")
 500                    rows.append(np.asarray(mz_cal_profile))
 501
 502                mz_abun_array = np.array(rows, dtype=profile_dtype)
 503
 504                raw_ms_dataset = scan_group.create_dataset(
 505                    "raw_ms",
 506                    data=mz_abun_array,
 507                    dtype=profile_dtype,
 508                    **hdf5_dataset_kwargs(mz_abun_array.shape, compression=compression),
 509                )
 510                raw_ms_dataset.attrs["row_labels"] = json.dumps(row_labels)
 511                # Recorded independently of whether an "mz_cal" row was
 512                # written, so a reader can trust this over row presence.
 513                raw_ms_dataset.attrs["is_calibrated"] = bool(
 514                    mass_spectrum.is_calibrated
 515                )
 516
 517                # Add metadata to the raw_ms dataset when it exists
 518                raw_ms_dataset.attrs["MassSpecAttrs"] = json.dumps(dict_ms_attrs)
 519
 520                if isinstance(mass_spectrum, MassSpecfromFreq):
 521                    raw_ms_dataset.attrs["TransientSetting"] = json.dumps(
 522                        setting_dicts.get("TransientSetting"),
 523                        sort_keys=False,
 524                        indent=4,
 525                        separators=(",", ": "),
 526                    )
 527            else:
 528                # For centroided data, store metadata directly on the scan group
 529                scan_group.attrs["MassSpecAttrs"] = json.dumps(dict_ms_attrs)
 530
 531                if isinstance(mass_spectrum, MassSpecfromFreq):
 532                    scan_group.attrs["TransientSetting"] = json.dumps(
 533                        setting_dicts.get("TransientSetting"),
 534                        sort_keys=False,
 535                        indent=4,
 536                        separators=(",", ": "),
 537                    )
 538
 539        else:
 540            scan_group = hdf_handle.get(group_key)
 541
 542        # Next processed-data index: one past the highest existing numeric
 543        # key, or 0 if there are none. "raw_ms" and "peak_profile_indices_*"
 544        # are not processed-data indices (get_time_index_to_pull also only
 545        # considers numeric keys, for the same reason). Using the max rather
 546        # than a count matters for a legacy file that already has "raw_ms"
 547        # and "1" (an old off-by-one: pre-fix code counted "raw_ms" as an
 548        # occupied slot) -- counting numeric keys would give "1" again and
 549        # collide with the existing one; max+1 gives "2".
 550        numeric_keys = [int(k) for k in scan_group.keys() if k.isdigit()]
 551        index_processed_data = str(max(numeric_keys) + 1) if numeric_keys else "0"
 552
 553        # Per-peak (left, apex, right) indices into the profile arrays above,
 554        # in mspeaks order, so a reloaded centroid-with-profile mass
 555        # spectrum doesn't need to re-derive peak boundaries from the peak
 556        # table's row position (which isn't a profile index). Written once
 557        # per peak-table version (same suffix as the processed-data index
 558        # below), since the profile itself (raw_ms) doesn't change across
 559        # reprocessing but the picked peaks do -- except raw_ms itself is
 560        # only ever written on the group's first creation (just above), so
 561        # if *this* mass_spectrum's profile is a different length (e.g. a
 562        # later reprocessing that re-picked over different raw data), these
 563        # indices would point into a profile that was never stored. Skip
 564        # writing them in that case; the reader falls back to a search.
 565        existing_raw_ms = scan_group.get("raw_ms")
 566        if (
 567            not mass_spectrum.is_centroid
 568            and export_raw
 569            and existing_raw_ms is not None
 570            and existing_raw_ms.shape[-1] == len(mass_spectrum.abundance_profile)
 571        ):
 572            peak_indices = np.array(
 573                [
 574                    (
 575                        mspeak.peak_left_index,
 576                        mspeak.peak_apex_index,
 577                        mspeak.peak_right_index,
 578                    )
 579                    for mspeak in mass_spectrum.mspeaks
 580                ],
 581                dtype=np.int32,
 582            ).reshape(-1, 3)
 583            scan_group.create_dataset(
 584                f"peak_profile_indices_{index_processed_data}",
 585                data=peak_indices,
 586                **hdf5_dataset_kwargs(peak_indices.shape, compression=compression),
 587            )
 588
 589        timenow = str(datetime.now(timezone.utc).strftime("%d/%m/%Y %H:%M:%S %Z"))
 590
 591        # Convert list_results to numpy array and optimize data types
 592        array_results = np.array(list_results)
 593
 594        # Apply data type optimization for numeric columns
 595        if array_results.dtype == np.float64:
 596            array_results = array_results.astype(np.float32)
 597        elif array_results.dtype == np.int64:
 598            array_results = array_results.astype(np.int32)
 599
 600        processed_dset = scan_group.create_dataset(
 601            index_processed_data,
 602            data=array_results,
 603            **hdf5_dataset_kwargs(array_results.shape, compression=compression),
 604        )
 605
 606        processed_dset.attrs["date_utc"] = timenow
 607
 608        processed_dset.attrs["ColumnsLabels"] = columns_labels
 609
 610        processed_dset.attrs["MoleculaSearchSetting"] = json.dumps(
 611            setting_dicts.get("MoleculaSearch"),
 612            sort_keys=False,
 613            indent=4,
 614            separators=(",", ": "),
 615        )
 616
 617        processed_dset.attrs["MassSpecPeakSetting"] = json.dumps(
 618            setting_dicts.get("MassSpecPeak"),
 619            sort_keys=False,
 620            indent=4,
 621            separators=(",", ": "),
 622        )
 623
 624        processed_dset.attrs["MassSpectrumSetting"] = json.dumps(
 625            setting_dicts.get("MassSpectrum"),
 626            sort_keys=False,
 627            indent=4,
 628            separators=(",", ": "),
 629        )
