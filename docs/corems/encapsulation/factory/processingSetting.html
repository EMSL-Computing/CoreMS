  26def settings_to_export_dict(settings_obj) -> dict:
  27    """Serialize settings for parameter export; skip fields marked legacy.
  28
  29    Fields defined with :func:`legacy_field` (metadata
  30    ``LEGACY_PARAMETER=True``) are omitted. Used by LCMS parameter dump so
  31    old annotation knobs on ``LiquidChromatographSetting`` are not rewritten
  32    into TOML/JSON after migration to ``spectral_similarity_search``.
  33    """
  34    if not dataclasses.is_dataclass(settings_obj) or isinstance(settings_obj, type):

  44def validate_used_atoms_keys(used_atoms):
  45    """Validate that usedAtoms keys are element symbols, not rare isotopes.
  46
  47    Molecular formula search treats each usedAtoms key as a monoisotopic
  48    (most-abundant) element. Heavy isotopologues (e.g. 13C, 54Fe, 37Cl) are
  49    produced later by isotopologue expansion of mono formulas, not by listing
  50    rare-isotope labels in usedAtoms. Membership in ``Atoms.isotopes`` does
  51    not mean the element is ready to search: each key still needs a valence
  52    in ``used_atom_valences`` (typically copied from ``Atoms.atoms_covalence``),
  53    or DBE calculation can fail.
  54
  55    Parameters
  56    ----------
  57    used_atoms : dict
  58        Mapping of atom symbol to (min, max) count range.
  59
  60    Raises
  61    ------
  62    ValueError
  63        If any key is not a mono element symbol (key of Atoms.isotopes).
  64    TypeError
  65        If used_atoms is not a mapping.
  66    """
  67    if used_atoms is None:
  68        return
  69    if not hasattr(used_atoms, "keys"):
  70        raise TypeError(
  71            "usedAtoms must be a dict of element symbols to (min, max) ranges"
  72        )
  73    invalid = [k for k in used_atoms.keys() if k not in Atoms.isotopes]
  74    if invalid:
  75        raise ValueError(
  76            "usedAtoms keys must be element symbols (e.g. Fe, C, Cl), "
  77            "not specific isotope labels (e.g. 54Fe, 13C, 37Cl). "
  78            f"Invalid key(s): {invalid}. "
  79            "Heavy isotopologues are generated from mono formulas during search."
  80        )
